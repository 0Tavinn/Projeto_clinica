def leak_forms(value):

  before=db.scalar(text("SELECT count(*) FROM clinical_evolutions"))
  db.execute(text("CREATE FUNCTION qa_n05_failure() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN RAISE EXCEPTION 'QA26_EXCEPTION_5129 %', NEW.description; END $$"))
  db.execute(text("CREATE TRIGGER qa_n05_failure BEFORE INSERT ON clinical_evolutions FOR EACH ROW EXECUTE FUNCTION qa_n05_failure()"))

  r=request("POST","/api/v1/evolutions?q="+quote(MARKERS[0],safe=""),clinical,{"patient_id":p,"description":MARKERS[1],"notes":MARKERS[2]})

  with E.begin() as db:
   db.execute(text("DROP TRIGGER qa_n05_failure ON clinical_evolutions"));db.execute(text("DROP FUNCTION qa_n05_failure()"))
